import { RegisterClient } from './application/use-cases/auth/RegisterClient'
import { GetLocations } from './application/use-cases/location/GetLocations'

// Creamos las dependencias una sola vez.
const authRepository = new HttpAuthRepository()
const locationRepository = new HttpLocationRepository()
const sessionRepository = new LocalStorageSessionRepository()

const loginUseCase = new Login(authRepository)
const registerClientUseCase = new RegisterClient(authRepository)
const getLocationsUseCase = new GetLocations(locationRepository)

import './presentation/styles/auth.css'

/**
 * Contiene la interfaz y la lógica de autenticación.
 * Se renderiza dentro de BrowserRouter y AuthSessionProvider.
 */
function AppContent() {
  const navigate = useNavigate()
  const { setSession } = useAuthSession()

  // Gestiona el formulario de login y registro.
  const {
    authMode,
    form,
    message,
    isSubmitting,
    registrationCompleted,
    locations,
    isLoadingLocations,
    locationsError,
    handleFormChange,
    openAuth,
    closeAuth,
    handleSubmit,
    handleGoToPanel,
  } = useAuth({
    loginUseCase,
    registerClientUseCase,
    getLocationsUseCase,

    // Guarda la sesión y redirige según el rol.
    onLoginSuccess: (session) => {
      setSession(session)
      console.log('Usuario autenticado:', session.user)
      console.log('Rol recibido:', session.user.role)
      navigate(getRolePath(session.user.role), {
        replace: true,
      })
    },
  })

  return (
    <AppRouter
      homePage={
        <>
          <LandingPage onOpenAuth={openAuth} />

          {authMode && (
            <AuthModal
              mode={authMode}
              form={form}
              locations={locations}
              isLoadingLocations={isLoadingLocations}
              locationsError={locationsError}
              message={message}
              isSubmitting={isSubmitting}
              registrationCompleted={registrationCompleted}
              onClose={closeAuth}
              onSubmit={handleSubmit}
              onGoToPanel={handleGoToPanel}
              onContinueLater={closeAuth}
              onSwitchMode={() =>
                openAuth(authMode === 'register' ? 'login' : 'register')
              }
              onFormChange={handleFormChange}
            />
          )}
        </>
      }
    />
  )
}

/**
 * Componente raíz de la aplicación.
 * Proporciona el enrutador y el contexto de autenticación.
 */
function App() {
  return (
    <BrowserRouter>
      <AuthSessionProvider sessionRepository={sessionRepository}>
        <AppContent />
      </AuthSessionProvider>
    </BrowserRouter>
  )
}

export default App

apiClient.interceptors.request.use((config) => {
  const storedSession = localStorage.getItem(AUTH_SESSION_KEY)

  if (storedSession) {
    try {
      const session = JSON.parse(storedSession) as { token: string }

      config.headers.Authorization = `Bearer ${session.token}`
    } catch (error) {
      console.error('No se pudo recuperar el token:', error)
    }
  }

  return config
})

import type { SessionRepository } from '../../application/ports/SessionRepository'
import type { AuthSession } from '../../application/dtos/auth'

// Clave única utilizada para guardar la sesión.
export const AUTH_SESSION_KEY = 'matcom_auth_session'

/**
 * Implementa la persistencia de la sesión utilizando localStorage.
 */
export class LocalStorageSessionRepository implements SessionRepository {
  /**
   * Recupera la sesión almacenada.
   * Si no existe o está dañada, devuelve null.
   */
  getSession(): AuthSession | null {
    try {
      const storedSession = localStorage.getItem(AUTH_SESSION_KEY)

      if (!storedSession) {
        return null
      }

      return JSON.parse(storedSession) as AuthSession
    } catch (error) {
      console.error('No se pudo recuperar la sesión:', error)

      this.clearSession()
      return null
    }
  }

  /**
   * Guarda el token y los datos del usuario.
   */
  saveSession(session: AuthSession): void {
    localStorage.setItem(AUTH_SESSION_KEY, JSON.stringify(session))
  }

  /**
   * Elimina la sesión almacenada.
   */
  clearSession(): void {
    localStorage.removeItem(AUTH_SESSION_KEY)
  }
}

type AgeInputProps = {

function AgeInput({ value, onChange }: AgeInputProps) {
  const age = Number(value)

  // Incrementa la edad sin superar el máximo permitido.
  function increaseAge() {
    onChange(String(Math.min(120, (age || 0) + 1)))
  }

  // Disminuye la edad sin bajar de 1.
  function decreaseAge() {
    onChange(String(Math.max(1, (age || 2) - 1)))
  }

  return (
    <div className="age-control">
      <input
        className="age-input"
        type="number"
        name="age"
        min={1}
        max={120}
        step={1}
        value={value}

import { useEffect, useRef, useState } from 'react'
import type { KeyboardEvent } from 'react'

// Define la estructura de cada opción del selector.
export type SelectOption = {
  value: string
  label: string
}

// Propiedades que recibe el componente.
type CustomSelectProps = {
  value: string
  options: SelectOption[]
  placeholder: string
  onChange: (value: string) => void
  disabled?: boolean
}

function CustomSelect({
  value,
  options,
  placeholder,
  onChange,
  disabled = false,
}: CustomSelectProps) {
  const [isOpen, setIsOpen] = useState(false)
  const [activeIndex, setActiveIndex] = useState(0)

  const containerRef = useRef<HTMLDivElement>(null)

  // Busca la opción que coincide con el valor seleccionado.
  const selectedOption = options.find(
    (option) => option.value === value,
  )

  // Cierra el menú cuando se hace clic fuera del selector.
  useEffect(() => {
    function handleOutsideClick(event: PointerEvent) {
      if (
        containerRef.current &&
        !containerRef.current.contains(event.target as Node)
      ) {
        setIsOpen(false)
      }
    }

    document.addEventListener('pointerdown', handleOutsideClick)

    return () => {
      document.removeEventListener('pointerdown', handleOutsideClick)
    }
  }, [])

  // Selecciona una opción y cierra el menú.
  function chooseOption(option: SelectOption) {
    onChange(option.value)
    setIsOpen(false)
  }

  // Gestiona la navegación mediante el teclado.
  function handleKeyDown(event: KeyboardEvent<HTMLButtonElement>) {
    if (disabled || options.length === 0) return

    if (event.key === 'ArrowDown' || event.key === 'ArrowUp') {
      event.preventDefault()

      if (!isOpen) {
        setIsOpen(true)

        const selectedIndex = options.findIndex(
          (option) => option.value === value,
        )

        setActiveIndex(selectedIndex >= 0 ? selectedIndex : 0)
        return
      }

      setActiveIndex((current) => {
        if (event.key === 'ArrowDown') {
          return (current + 1) % options.length
        }

        return (current - 1 + options.length) % options.length
      })
    }

    if (event.key === 'Enter' && isOpen) {
      event.preventDefault()
      chooseOption(options[activeIndex])
    }

    if (event.key === 'Escape') {
      setIsOpen(false)
    }

    if (event.key === 'Home' && isOpen) {
      event.preventDefault()
      setActiveIndex(0)
    }

    if (event.key === 'End' && isOpen) {
      event.preventDefault()
      setActiveIndex(options.length - 1)
    }
  }

  return (
    <div className="custom-select" ref={containerRef}>
      <button
        type="button"
        className={`custom-select-trigger ${isOpen ? 'is-open' : ''}`}
        aria-haspopup="listbox"
        aria-expanded={isOpen}
        disabled={disabled}
        onClick={() => {
          if (disabled || options.length === 0) return

          if (!isOpen) {
            const selectedIndex = options.findIndex(
              (option) => option.value === value,
            )

            setActiveIndex(selectedIndex >= 0 ? selectedIndex : 0)
          }

          setIsOpen((open) => !open)
        }}
        onKeyDown={handleKeyDown}
      >
        <span className={!selectedOption ? 'custom-select-placeholder' : ''}>
          {selectedOption?.label ?? placeholder}
        </span>

        <svg
          className="custom-select-chevron"
          viewBox="0 0 24 24"
          width="18"
          height="18"
          fill="none"
          aria-hidden="true"
        >
          <path
            d="m6 9 6 6 6-6"
            stroke="currentColor"
            strokeWidth="2"
            strokeLinecap="round"

    return <Navigate to={APP_PATHS.HOME} replace />

        replace
      />
    )
  }

  return <Outlet />

  function setSession(newSession: AuthSession) {
    sessionRepository.saveSession(newSession)
    setCurrentSession(newSession)
  }

  /**
   * Cierra la sesión y elimina los datos almacenados.
   */
  function logout() {
    sessionRepository.clearSession()
    setCurrentSession(null)
  }

  return (
    <AuthSessionContext.Provider value={{ session, setSession, logout }}>
      {children}
    </AuthSessionContext.Provider>
  )

export function useAuth({
  loginUseCase,
  registerClientUseCase,
  getLocationsUseCase,
  onLoginSuccess,
}: UseAuthDependencies) {
  const [authMode, setAuthMode] = useState<AuthMode | null>(null)
  const [form, setForm] = useState<AuthForm>({ ...INITIAL_FORM })
  const [message, setMessage] = useState('')
  const [isSubmitting, setIsSubmitting] = useState(false)
  const [registrationCompleted, setRegistrationCompleted] = useState(false)

  const [locations, setLocations] = useState<Location[]>([])
  const [isLoadingLocations, setIsLoadingLocations] = useState(false)
  const [locationsError, setLocationsError] = useState('')

  useEffect(() => {
    async function loadLocations() {
      try {
        setIsLoadingLocations(true)
        const data = await getLocationsUseCase.execute()
        setLocations(data)
      } catch (error) {
        console.error('Error al cargar las sedes:', error)
        setLocationsError(
          'No se pudieron cargar las sedes. Recarga la página.',
        )
      } finally {
        setIsLoadingLocations(false)
      }
    }

    loadLocations()
  }, [getLocationsUseCase])

  function handleFormChange(field: keyof AuthForm, value: string) {
    setForm((currentForm) => ({
      ...currentForm,
      [field]: value,
    }))
  }

  function openAuth(mode: AuthMode) {
    setAuthMode(mode)
    setForm({ ...INITIAL_FORM })
    setMessage('')
    setRegistrationCompleted(false)
  }

  function closeAuth() {
    setAuthMode(null)
    setMessage('')
    setRegistrationCompleted(false)
  }

  async function handleSubmit(event: FormEvent<HTMLFormElement>) {
    event.preventDefault()
    setMessage('')

    if (authMode === 'register' && !form.gender) {
      setMessage('Selecciona una opción de género.')
      return
    }

    if (authMode === 'register' && !form.idLocation) {
      setMessage('Selecciona una sede del gimnasio.')
      return
    }

    setIsSubmitting(true)

    try {
      if (authMode === 'login') {
        const session = await loginUseCase.execute({
          email: form.email.trim(),
          password: form.password,
        })

        onLoginSuccess(session)
        closeAuth()
      } else if (authMode === 'register') {
        await registerClientUseCase.execute({
          name: form.name.trim(),
          email: form.email.trim(),
          phone: form.phone.trim(),
          password: form.password,
          role: 'CLIENT',
          age: Number(form.age),
          gender: form.gender as 'FEMENINO' | 'MASCULINO' | 'OTRO',
          idLocation: Number(form.idLocation),
        })

        setRegistrationCompleted(true)
      }
    } catch (error: unknown) {
      console.error('Error durante la autenticación:', error)

      if (axios.isAxiosError(error)) {
        console.error('Respuesta del servidor:', error.response?.data)
        console.error('Código HTTP:', error.response?.status)

        setMessage(
          error.response?.data?.error?.message ??
            'No se pudo completar la operación.',
        )
      } else if (error instanceof Error) {
        console.error('Mensaje:', error.message)
        setMessage(error.message)
      } else {
        setMessage('Ocurrió un error inesperado.')
      }
    } finally {
      setIsSubmitting(false)
    }
  }

  // Inicia sesión después de que el usuario elija ir a su panel.
  async function handleGoToPanel() {
    setIsSubmitting(true)
    setMessage('')

    try {
      const session = await loginUseCase.execute({
        email: form.email.trim(),
        password: form.password,
      })

      onLoginSuccess(session)
      closeAuth()
    } catch (error) {
      console.error('Error al iniciar sesión automáticamente:', error)

      setMessage(
        'Tu cuenta fue creada, pero no pudimos iniciar sesión automáticamente. Inténtalo desde el inicio de sesión.',
      )
    } finally {
      setIsSubmitting(false)
    }
  }

  return {
    authMode,
    form,
    message,
    isSubmitting,
    registrationCompleted,
    locations,
    isLoadingLocations,
    locationsError,
    handleFormChange,
    openAuth,
    closeAuth,
    handleSubmit,
    handleGoToPanel,
  }
}
